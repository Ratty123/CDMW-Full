cmake_minimum_required(VERSION 3.24)
project(cdmw_virtual_loader LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The virtual loader requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(cdmw_virtual_protocol STATIC protocol.cpp)
target_compile_definitions(cdmw_virtual_protocol PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(cdmw_virtual_protocol PUBLIC bcrypt advapi32)
add_library(cdmw_virtual_loader SHARED runtime.cpp)
target_link_libraries(cdmw_virtual_loader PRIVATE cdmw_virtual_protocol)
set_target_properties(cdmw_virtual_loader PROPERTIES PREFIX "" OUTPUT_NAME "CDMWVirtualLoader" SUFFIX ".asi")
add_executable(cdmw_virtual_session session.cpp)
target_link_libraries(cdmw_virtual_session PRIVATE cdmw_virtual_protocol)
add_executable(cdmw_virtual_loader_harness harness.cpp)
target_link_libraries(cdmw_virtual_loader_harness PRIVATE cdmw_virtual_protocol)
enable_testing()
add_test(NAME protocol_and_io COMMAND cdmw_virtual_loader_harness --self-test)
foreach(target cdmw_virtual_protocol cdmw_virtual_loader cdmw_virtual_session cdmw_virtual_loader_harness)
  target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc /guard:cf)
endforeach()
