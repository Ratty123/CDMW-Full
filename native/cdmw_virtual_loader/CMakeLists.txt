cmake_minimum_required(VERSION 3.24)
project(cdmw_virtual_loader LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The virtual loader requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
foreach(field CDMW_APP_VERSION CDMW_FILE_VERSION CDMW_PRODUCT_NAME CDMW_COMPANY_NAME CDMW_REPOSITORY_URL)
  if(NOT DEFINED ${field} OR "${${field}}" STREQUAL "")
    message(FATAL_ERROR "Configure with scripts/build_virtual_loader.py to supply application version metadata")
  endif()
endforeach()
set(CDMW_VERSION_FLAGS "0")
if(CDMW_APP_VERSION MATCHES "-")
  set(CDMW_VERSION_FLAGS "VS_FF_PRERELEASE")
endif()
function(add_version_resource target description filename file_type)
  set(CDMW_DESCRIPTION "${description}")
  set(CDMW_ORIGINAL_FILENAME "${filename}")
  string(REGEX REPLACE "\\.[^.]+$" "" CDMW_INTERNAL_NAME "${filename}")
  set(CDMW_FILE_TYPE "${file_type}")
  configure_file(version.rc.in "${target}_version.rc" @ONLY)
  target_sources(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/${target}_version.rc")
endfunction()

add_library(cdmw_virtual_protocol STATIC protocol.cpp)
add_library(cdmw_virtual_protocol_test STATIC protocol.cpp)
target_compile_definitions(cdmw_virtual_protocol_test PUBLIC CDMW_VIRTUAL_TESTING=1)
foreach(target cdmw_virtual_protocol cdmw_virtual_protocol_test)
  target_compile_definitions(${target} PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(${target} PUBLIC bcrypt advapi32)
endforeach()
# The launch-time composition links the portable item composer.
add_subdirectory(../cdmw_item_composer "${CMAKE_CURRENT_BINARY_DIR}/item_composer")
add_library(cdmw_virtual_loader SHARED runtime.cpp bootstrap.cpp compose_launch.cpp)
target_link_libraries(cdmw_virtual_loader PRIVATE cdmw_virtual_protocol cdmw_item_composer)
set_target_properties(cdmw_virtual_loader PROPERTIES PREFIX "" OUTPUT_NAME "CDMWVirtualLoader" SUFFIX ".asi")
add_version_resource(cdmw_virtual_loader "CDMW Virtual Loader" "CDMWVirtualLoader.asi" VFT_DLL)
add_executable(cdmw_virtual_session session.cpp)
target_link_libraries(cdmw_virtual_session PRIVATE cdmw_virtual_protocol)
add_version_resource(cdmw_virtual_session "CDMW Virtual Session" "cdmw_virtual_session.exe" VFT_APP)

# Test-mode launch and activation code is compiled only into these local fixtures.
add_library(cdmw_virtual_loader_test SHARED runtime.cpp bootstrap.cpp compose_launch.cpp)
target_link_libraries(cdmw_virtual_loader_test PRIVATE cdmw_virtual_protocol_test cdmw_item_composer)
set_target_properties(cdmw_virtual_loader_test PROPERTIES PREFIX "" OUTPUT_NAME "CDMWVirtualLoaderTest" SUFFIX ".asi")
add_version_resource(cdmw_virtual_loader_test "CDMW Virtual Loader Test Fixture" "CDMWVirtualLoaderTest.asi" VFT_DLL)
add_executable(cdmw_virtual_session_test session.cpp)
target_link_libraries(cdmw_virtual_session_test PRIVATE cdmw_virtual_protocol_test)
add_version_resource(cdmw_virtual_session_test "CDMW Virtual Session Test Fixture" "cdmw_virtual_session_test.exe" VFT_APP)
add_executable(cdmw_virtual_loader_harness harness.cpp bootstrap.cpp)
target_link_libraries(cdmw_virtual_loader_harness PRIVATE cdmw_virtual_protocol_test)
set_target_properties(cdmw_virtual_protocol_test cdmw_virtual_loader_test cdmw_virtual_session_test cdmw_virtual_loader_harness
  PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/tests"
             LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/tests"
             ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/tests")
enable_testing()
add_test(NAME protocol_and_io COMMAND cdmw_virtual_loader_harness --self-test)
foreach(target cdmw_virtual_protocol cdmw_virtual_loader cdmw_virtual_session
               cdmw_virtual_protocol_test cdmw_virtual_loader_test cdmw_virtual_session_test cdmw_virtual_loader_harness)
  target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/WX;/permissive-;/EHsc;/guard:cf>")
endforeach()
